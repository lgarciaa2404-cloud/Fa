--- 1B. Basesita.java (crea la BD y hace altas, bajas, cambios, consultas) ---

CAMBIAR: nombre de la tabla "empleaditos", sus columnas y tipos
(int, text, real) en el create table, y los mismos nombres en los
ContentValues y en los select.
El nombre de la BD ("administra") se puede poner en el constructor.

package com.example.faahh;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

import androidx.annotation.Nullable;

import java.util.ArrayList;

public class Basesita extends SQLiteOpenHelper {

    public Basesita(@Nullable Context context) {
        super(context, "administra", null, 1);
    }

    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL("create table empleaditos(clave int primary key, puesto text, depto text, sueldo int)");
    }

    @Override
    public void onUpgrade(SQLiteDatabase db, int oldVersion, int newVersion) {
    }

    // ALTAS: true si se guardo, false si la clave ya existe
    public boolean altas(Datitos d) {
        SQLiteDatabase db = getWritableDatabase();
        ContentValues registro = new ContentValues();
        registro.put("clave", d.getClave());
        registro.put("puesto", d.getPuesto());
        registro.put("depto", d.getDepto());
        registro.put("sueldo", d.getSueldo());
        return db.insert("empleaditos", null, registro) != -1;
    }

    // BAJAS: cuantos borro (0 = no existia)
    public int bajas(int clave) {
        SQLiteDatabase db = getWritableDatabase();
        return db.delete("empleaditos", "clave=" + clave, null);
    }

    // CAMBIOS: cuantos modifico (0 = no existia)
    public int cambios(Datitos d) {
        SQLiteDatabase db = getWritableDatabase();
        ContentValues registro = new ContentValues();
        registro.put("puesto", d.getPuesto());
        registro.put("depto", d.getDepto());
        registro.put("sueldo", d.getSueldo());
        return db.update("empleaditos", registro, "clave=" + d.getClave(), null);
    }

    // CONSULTAS: busca por clave (null si no existe)
    public Datitos consultas(int clave) {
        SQLiteDatabase db = getReadableDatabase();
        Cursor fila = db.rawQuery(
                "select clave, puesto, depto, sueldo from empleaditos where clave=" + clave, null);
        Datitos d = null;
        if (fila.moveToFirst()) {
            d = new Datitos(fila.getInt(0), fila.getString(1), fila.getString(2), fila.getInt(3));
        }
        fila.close();
        return d;
    }

    // LISTAR: todos los registros
    public ArrayList<Datitos> listar() {
        SQLiteDatabase db = getReadableDatabase();
        Cursor fila = db.rawQuery(
                "select clave, puesto, depto, sueldo from empleaditos order by clave", null);
        ArrayList<Datitos> lista = new ArrayList<>();
        while (fila.moveToNext()) {
            lista.add(new Datitos(fila.getInt(0), fila.getString(1), fila.getString(2), fila.getInt(3)));
        }
        fila.close();
        return lista;
    }
}

package com.example.faahh;

import android.os.Bundle;
import android.view.MenuItem;

import androidx.activity.OnBackPressedCallback;
import androidx.annotation.NonNull;
import androidx.appcompat.app.ActionBarDrawerToggle;
import androidx.appcompat.app.AppCompatActivity;
import androidx.appcompat.widget.Toolbar;
import androidx.core.view.GravityCompat;
import androidx.drawerlayout.widget.DrawerLayout;
import androidx.fragment.app.Fragment;

import com.google.android.material.navigation.NavigationView;

public class MainActivity extends AppCompatActivity
        implements NavigationView.OnNavigationItemSelectedListener {

    DrawerLayout drawerLayout;
    NavigationView navigationView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        Toolbar toolbar = findViewById(R.id.toolbar);
        setSupportActionBar(toolbar);

        drawerLayout = findViewById(R.id.drawer_layout);
        navigationView = findViewById(R.id.nav_view);
        navigationView.setNavigationItemSelectedListener(this);

        ActionBarDrawerToggle toggle = new ActionBarDrawerToggle(this,
                drawerLayout, toolbar, R.string.open_nav, R.string.close_nav);
        drawerLayout.addDrawerListener(toggle);
        toggle.syncState();

        // Boton atras: si el menu esta abierto, lo cierra
        getOnBackPressedDispatcher().addCallback(this, new OnBackPressedCallback(true) {
            @Override
            public void handleOnBackPressed() {
                if (drawerLayout.isDrawerOpen(GravityCompat.START)) {
                    drawerLayout.closeDrawer(GravityCompat.START);
                } else {
                    setEnabled(false);
                    getOnBackPressedDispatcher().onBackPressed();
                }
            }
        });

        // Al abrir la app muestra el primer fragment
        if (savedInstanceState == null) {
            mostrar(new UnoFragment());
            navigationView.setCheckedItem(R.id.nav_uno);
        }
    }

    // Se llama cuando tocas una opcion del menu lateral
    @Override
    public boolean onNavigationItemSelected(@NonNull MenuItem item) {
        int id = item.getItemId();

        if (id == R.id.nav_uno) {
            mostrar(new UnoFragment());
        } else if (id == R.id.nav_dos) {
            mostrar(new DosFragment());
        } else if (id == R.id.nav_tres) {
            mostrar(new TresFragment());
        }

        drawerLayout.closeDrawer(GravityCompat.START);
        return true;
    }

    // Cambia el fragment que se ve. Tambien se puede llamar desde un fragment:
    //   ((MainActivity) requireActivity()).mostrar(new UnoFragment());
    public void mostrar(Fragment fragment) {
        getSupportFragmentManager()
                .beginTransaction()
                .replace(R.id.fragment_container, fragment)
                .commit();
    }
}

--- 4B. UnoFragment.java ---

CAMBIAR: los campos (EditText) y los datos que se leen. Si llega un
dato desde otro fragment (clave) lo carga automaticamente.

package com.example.faahh;

import android.os.Bundle;
import android.view.View;
import android.widget.EditText;
import android.widget.Toast;

import androidx.annotation.NonNull;
import androidx.fragment.app.Fragment;

public class UnoFragment extends Fragment implements View.OnClickListener {

    EditText eclave, epuesto, edepto, esueldo;
    Basesita admin;

    public UnoFragment() {
        super(R.layout.fragment_uno);
    }

    @Override
    public void onViewCreated(@NonNull View view, Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        eclave = view.findViewById(R.id.eclave);
        epuesto = view.findViewById(R.id.epuesto);
        edepto = view.findViewById(R.id.edepto);
        esueldo = view.findViewById(R.id.esueldo);

        admin = new Basesita(requireContext());

        view.findViewById(R.id.altas).setOnClickListener(this);
        view.findViewById(R.id.bajas).setOnClickListener(this);
        view.findViewById(R.id.cambios).setOnClickListener(this);
        view.findViewById(R.id.consultas).setOnClickListener(this);

        // Dato que manda otro fragment (ver seccion 5)
        Bundle args = getArguments();
        if (args != null && args.containsKey("clave")) {
            Datitos d = admin.consultas(args.getInt("clave"));
            if (d != null) ponerEnCampos(d);
        }
    }

    @Override
    public void onClick(View v) {
        int id = v.getId();

        if (id == R.id.altas) {
            Datitos d = leerCampos();
            if (d == null) return;
            if (admin.altas(d)) {
                aviso("agregado");
                limpiar();
            } else {
                aviso("Ya existe esa clave");
            }

        } else if (id == R.id.bajas) {
            int clave = leerClave();
            if (clave < 0) return;
            int borrados = admin.bajas(clave);
            aviso(borrados > 0 ? "eliminado" : "No existe esa clave");
            limpiar();

        } else if (id == R.id.cambios) {
            Datitos d = leerCampos();
            if (d == null) return;
            int modificados = admin.cambios(d);
            aviso(modificados > 0 ? "modificado" : "No existe esa clave");
            if (modificados > 0) limpiar();

        } else if (id == R.id.consultas) {
            int clave = leerClave();
            if (clave < 0) return;
            Datitos d = admin.consultas(clave);
            if (d == null) {
                aviso("No existe esa clave");
            } else {
                ponerEnCampos(d);
            }
        }
    }

    private void aviso(String texto) {
        Toast.makeText(requireContext(), texto, Toast.LENGTH_SHORT).show();
    }

    private void ponerEnCampos(Datitos d) {
        eclave.setText(String.valueOf(d.getClave()));
        epuesto.setText(d.getPuesto());
        edepto.setText(d.getDepto());
        esueldo.setText(String.valueOf(d.getSueldo()));
    }

    private void limpiar() {
        eclave.setText("");
        epuesto.setText("");
        edepto.setText("");
        esueldo.setText("");
    }

    // Devuelve la clave escrita, o -1 si esta vacia o es invalida
    private int leerClave() {
        String c = eclave.getText().toString().trim();
        if (c.isEmpty()) {
            aviso("Escribe la clave");
            return -1;
        }
        try {
            return Integer.parseInt(c);
        } catch (NumberFormatException e) {
            aviso("Clave invalida");
            return -1;
        }
    }

    // Lee y valida todos los campos; null si algo esta mal
    private Datitos leerCampos() {
        int clave = leerClave();
        if (clave < 0) return null;

        String puesto = epuesto.getText().toString().trim();
        String depto = edepto.getText().toString().trim();
        String sueldoTxt = esueldo.getText().toString().trim();

        if (puesto.isEmpty() || depto.isEmpty() || sueldoTxt.isEmpty()) {
            aviso("Llena todos los campos");
            return null;
        }

        int sueldo;
        try {
            sueldo = Integer.parseInt(sueldoTxt);
        } catch (NumberFormatException e) {
            aviso("Sueldo invalido");
            return null;
        }

        return new Datitos(clave, puesto, depto, sueldo);
    }
}

--- 5B. DosFragment.java ---

Que hace:
  - Lee todos los registros de la BD y crea una fila por cada uno.
  - Clic en la fila: abre UnoFragment mandando la clave en un Bundle.
  - Clic en la celda depto: muestra un Toast (ejemplo de clic en celda).

CAMBIAR: las celdas (txt1...txt4) segun tus columnas y que hace el clic.

package com.example.faahh;

import android.os.Bundle;
import android.view.View;
import android.widget.TableLayout;
import android.widget.TableRow;
import android.widget.TextView;
import android.widget.Toast;

import androidx.annotation.NonNull;
import androidx.fragment.app.Fragment;

import java.util.ArrayList;

public class DosFragment extends Fragment {

    TableLayout tablita;
    Basesita admin;

    public DosFragment() {
        super(R.layout.fragment_dos);
    }

    @Override
    public void onViewCreated(@NonNull View view, Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        tablita = view.findViewById(R.id.tablita);
        admin = new Basesita(requireContext());

        ArrayList<Datitos> lista = admin.listar();

        for (final Datitos d : lista) {
            TableRow fila = new TableRow(requireContext());

            TextView txt1 = new TextView(requireContext());
            TextView txt2 = new TextView(requireContext());
            TextView txt3 = new TextView(requireContext());
            TextView txt4 = new TextView(requireContext());

            txt1.setPadding(10, 25, 10, 25);
            txt2.setPadding(10, 25, 10, 25);
            txt3.setPadding(10, 25, 10, 25);
            txt4.setPadding(10, 25, 10, 25);

            txt1.setText(String.valueOf(d.getClave()));
            txt2.setText(d.getPuesto());
            txt3.setText(d.getDepto());
            txt4.setText(String.valueOf(d.getSueldo()));

            // CLIC EN UNA CELDA: aqui, la celda depto
            txt3.setOnClickListener(new View.OnClickListener() {
                @Override
                public void onClick(View v) {
                    String depto = ((TextView) v).getText().toString();
                    Toast.makeText(requireContext(), "depto: " + depto, Toast.LENGTH_SHORT).show();
                }
            });

            // CLIC EN LA FILA: abre UnoFragment y le manda la clave (Bundle)
            fila.setOnClickListener(new View.OnClickListener() {
                @Override
                public void onClick(View v) {
                    Bundle datos = new Bundle();
                    datos.putInt("clave", d.getClave());

                    UnoFragment uno = new UnoFragment();
                    uno.setArguments(datos);

                    ((MainActivity) requireActivity()).mostrar(uno);
                }
            });

            fila.addView(txt1);
            fila.addView(txt2);
            fila.addView(txt3);
            fila.addView(txt4);
            tablita.addView(fila);
        }

        Toast.makeText(requireContext(), "son " + lista.size() + " registros", Toast.LENGTH_SHORT).show();
    }
}

Pasar datos entre fragments (resumen):
    Manda:    Bundle b = new Bundle();
              b.putInt("clave", 5);          // tambien putString("x", "hola")
              fragmentDestino.setArguments(b);
    Recibe:   Bundle args = getArguments();
              if (args != null && args.containsKey("clave")) {
                  int clave = args.getInt("clave");
              }

Nota: un clic en una celda NO pasa al clic de la fila (la celda lo
"consume"). Si una celda no tiene listener, el clic llega a la fila.

7. QUE DATOS LE METES TU (TABLA DE CAMBIOS)
=====================================================================

Si el examen cambia la tabla (ej. productos: codigo, nombre, precio):

  QUE CAMBIAR                         DONDE
  ----------------------------------  ------------------------------
  Nombre de la tabla y columnas       Basesita.onCreate (create table)
  Atributos de la clase de datos      Datitos.java (renombrar y/o clase)
  Nombres en registro.put("...")      Basesita (altas y cambios)
  Columnas en los select              Basesita (consultas y listar)
  EditText (uno por columna)          fragment_uno.xml
  Encabezado de la tabla              fragment_dos.xml
  Celdas txt1..txtN                   DosFragment (for)
  Opciones del menu lateral           menu_drawer.xml
  Un if por cada opcion               MainActivity.onNavigationItemSelected
  Una clase y un layout por pantalla  Un fragment por cada opcion

Tips para cambiar nombres rapido: Refactor -> Rename (Shift+F6) sobre
la clase o variable, y Ctrl+R (reemplazar) en el archivo.


=====================================================================
8. ERRORES COMUNES
=====================================================================

  - Rojo en todo R.id...: hay un error en algun XML. Corrige el XML,
    luego Build -> Clean Project -> Rebuild Project.
  - "no such table": desinstala la app del emulador y corre de nuevo.
  - "already has an action bar": el tema de la app no es NoActionBar.
  - El menu lateral abre pero no cambia el fragment: faltan los ifs
    en onNavigationItemSelected, o los ids del menu no coinciden.
  - NullPointerException en un fragment: findViewById se hizo sobre
    "view." (en Fragment se usa view.findViewById, no findViewById).
  - El nombre del archivo debe ser igual al de la clase
    (UnoFragment.java -> class UnoFragment).
  - Si no hay NavigationView/DrawerLayout reconocidos: estan en las
    librerias material y drawerlayout, que ya vienen en los proyectos
    nuevos. Si salen en rojo, Build -> Sync Project with Gradle Files.
